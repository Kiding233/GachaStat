**现状：** `_fit_local_polynomial` 使用局部逻辑回归在 200 个格点上估计光滑 P(失败|r) 曲线。存在四个理论缺陷——带宽选择无最优性（D）、无多重性校正（E）、格点噪声（F）、均匀带宽不适应数据密度（G）。

**新方案：三阶段管线**

```
资源数据 (r_i, y_i)
       │
       ▼
① binsglm 分位数分箱 ──── 只取 dots.fit + data_bin 边界
   │  IMSE 选 J（工程启发式，非理论最优）
   │  输出: x_left, x_right, p̂_j, N_j
       │
       ▼
② PAVA 保序估计 ──── 单调递减约束下合并逆向波动
   │  输出: θ̃_j, blocks, theta_tilde_per_bin
   │
   ├── 单调性诊断 ──── 检测策略断点（仅诊断，不触发算法回退）
   │   monotonicity_holds 存入返回字典供 GUI 状态栏展示
   │   <!-- REVIEW-R2-FIX: ISSUE-044 原架构图标注「若断点存在→局部回退独立检验+Sidak」为 v3 设计残留——实际代码中 _fit_vulnerability_pava 第438行 changepoint CI 无条件执行，monotonicity_holds 仅被 retreat_panel.py:347 读取用于状态栏文本展示，从未触发算法级行为变更。v3→v4 迭代中 Sidak 被变更点 CI 替代，但架构图和 GUI 警告文本未同步更新。 -->
   │
       ▼
③ 变更点估计 ──── ĵ* = max{j: θ̃_j > α}
   │  Bootstrap CI（固定箱边界）→ [ĵ*_lo, ĵ*_hi]
   │  输出: 脆弱区间右界（点估计 + 保守/乐观端）
```

**与旧方案（Wilson CI + Sidak）的核心区别：**

| | v3 混合方案 | v4 PAVA + 变更点 |
|---|---|---|
| 推断问题 | J 个独立检验 | 1 个变更点估计 |
| 多重性处理 | Sidak 校正 (O(J) 成本) | 变更点 CI (O(1) 成本) |
| 结构假设 | 无 | 单调递减（可检验、可回退） |
| 采样逆向波动 | 独立判定 → 可能断裂区间 | PAVA 自动吸收 → 区间强制连续 |
| 对 N_j 小的箱 | Wilson CI 变宽 → 丢失 | 被相邻箱合并 → 共享信息 |

```python
    """PAVA 保序回归。"""

    def test_pava_no_violation(self):
        """已递减数据不应被合并。"""

    def test_pava_merges_violators(self):
        """违反递减的相邻箱应被合并。"""
        from gacha_simulator.core.vulnerability import _pava
        p = [0.8, 0.65, 0.72, 0.45, 0.30]  # 箱 1→2 反了

        # 合并后的值应介于两者之间
        merged_vals = [theta[i] for i, blk in enumerate(blocks) if len(blk) > 1]

        # 输出应递减
        for i in range(len(theta) - 1):
```

```python
# gacha_simulator/core/vulnerability.py

def _pava(p, w):
    """Pool Adjacent Violators Algorithm——保序回归（递减约束）。

    Args:
        p: 原始比例估计 (length J)
        w: 权重（= N_j，每箱样本数）

    Returns:
        theta: 保序估计 (length L ≤ J)
        w_out: 块权重 (length L)
        blocks: 块结构 list of list，blocks[i] = 该块包含的原始箱索引
    """
    p = np.asarray(p, dtype=float)
    w = np.asarray(w, dtype=float)
    J = len(p)
    blocks = [[i] for i in range(J)]
    current_p = p.copy()
    current_w = w.copy()

    changed = True
    while changed:
        changed = False
        i = 1
        while i < len(current_p):
            if current_p[i - 1] < current_p[i]:  # 违反递减
                K_pool = (current_p[i - 1] * current_w[i - 1] +
                          current_p[i] * current_w[i])
                w_pool = current_w[i - 1] + current_w[i]
                <!-- REVIEW-FIX-PREV: ISSUE-024 零权重箱防护——若 w_pool < 1e-15（原始箱权重均为 0），
                     跳过合并直接删除当前箱，避免 K_pool/0 产生 NaN。
                     现实中 binsglm 分位数分箱保证 N_j > 0，但 _pava 作为通用工具需自防护。 -->
                if w_pool < 1e-15:
                    # 零权重箱——跳过合并，直接删除当前箱
                    current_p = np.delete(current_p, i)
                    current_w = np.delete(current_w, i)
                    del blocks[i]
                    changed = True
                    break  # 回溯
                current_p[i - 1] = K_pool / w_pool
                current_w[i - 1] = w_pool
                blocks[i - 1].extend(blocks[i])
                current_p = np.delete(current_p, i)
                current_w = np.delete(current_w, i)
                del blocks[i]
                changed = True
                break  # 回溯
            i += 1
    return current_p, current_w, blocks
```

```python
def _wilson_ci(n_success, n_total, level=0.95):
    """Wilson score 置信区间——不对 p̂=0 或 p̂=1 塌缩。"""
    from scipy.stats import norm
    if n_total == 0:
        return 0.0, 1.0
    z = norm.ppf(1 - (1 - level) / 2)
    p_hat_val = n_success / n_total
    denom = 1 + z**2 / n_total
    center = (p_hat_val + z**2 / (2 * n_total)) / denom
    margin = z * np.sqrt(
        p_hat_val * (1 - p_hat_val) / n_total + z**2 / (4 * n_total**2)
    ) / denom
    return center - margin, center + margin
```

```python
def _diagnose_monotonicity(p_hat, N_j, x_left, x_right, K_j=None):
    """诊断单调递减假设是否被策略断点违反。

    检查每对相邻箱的逆向波动幅度是否超过采样噪声能解释的范围。
    使用 Wilson SE（不对 p̂=0/1 塌缩），而非 Wald SE。

    阈值选择的统计依据：
    ─────────────────────────────
    将相邻箱差分标准化为 z_j = (p̂_{j+1} - p̂_j) / √(se_j² + se_{j+1}²)。
    在 H₀（单调递减成立）下，z_j 近似标准正态。对 J-1 个独立比较：

        Sidak 校正：α_per = 1 − (1 − α)^(1/(J-1))
        J=10 → z_{0.9972} ≈ 2.77
        J=15 → z_{0.9982} ≈ 2.91
        J=20 → z_{0.9987} ≈ 3.02
        J=30 → z_{0.9991} ≈ 3.13

    → 3σ ≈ Sidak 校正 5% FWER 当 J≈15–20（PAVA 后典型箱数）。
    → 1.5σ 为未校正宽松筛查线（per-comparison α≈0.13），仅标注「可疑」，
      不触发回退——避免对采样噪声过度反应。

    注意：z_j 仅近似正态——箱内二项比例 + Wilson CI 半宽近似 SE。
    小 N_j 或极端 p̂ 时近似质量下降，但 Wilson SE 比 Wald SE 退化慢得多。

    REVIEW-FIX-PREV: ISSUE-008
    注意：SE_j = (hi_j - lo_j) / (2 * 1.96) 是 Wilson CI 半宽的近似。
    Wilson score interval 在 p̂ 接近 0 或 1 时不对称——半宽除以标准正态分位数
    仅近似标准误，非精确值。对于 N_j 很小且 p̂ 极端的箱，此近似可能低估噪声尺度。
    若需要精确 SE，使用 Wilson 方差公式：
    var = p̂(1-p̂)/(n+z²) * n/(n+z²)。

    REVIEW-FIX-PREV: ISSUE-020
    若调用方传入 K_j（每箱成功计数），直接使用 K_j[j] 替代
    int(round(p_hat[j] * N_j[j])) 反算——消除浮点舍入风险。
    K_j 默认为 None 时回退到旧反算逻辑以保证向后兼容。

    Returns:
        breakpoints: list of dict(j, x_boundary, diff, noise_scale, severity)
        monotonicity_holds: bool —— True 表示无断点，单调性成立
    """
    # REVIEW-FIX-PREV: ISSUE-006 同模块内直接调用 _wilson_ci，无需 import
    J = len(p_hat)
    breakpoints = []

    for j in range(J - 1):
        if p_hat[j] < p_hat[j + 1]:  # 逆向波动
            diff = p_hat[j + 1] - p_hat[j]
            # Wilson SE：p̂=0 时也不塌缩
            # REVIEW-FIX-PREV: ISSUE-020 若 K_j 已传入则直接用，避免浮点反算
            if K_j is not None:
                k_j = K_j[j]
                k_j1 = K_j[j + 1]
            else:
                k_j = int(round(p_hat[j] * N_j[j]))
                k_j1 = int(round(p_hat[j + 1] * N_j[j + 1]))
            lo_j, hi_j = _wilson_ci(k_j, N_j[j], level=0.95)
            lo_j1, hi_j1 = _wilson_ci(k_j1, N_j[j + 1], level=0.95)
            # 用 Wilson CI 半宽近似 SE
            se_j = (hi_j - lo_j) / (2 * 1.96) if N_j[j] > 0 else 1.0
            se_j1 = (hi_j1 - lo_j1) / (2 * 1.96) if N_j[j + 1] > 0 else 1.0
            noise_scale = np.sqrt(se_j**2 + se_j1**2)

            if diff > 3 * noise_scale:
                severity = 'breakpoint'
            elif diff > 1.5 * noise_scale:
                severity = 'suspicious'
            else:
                severity = 'noise'

            breakpoints.append({
                'j': j,
                'x_boundary': x_left[j + 1],
                'diff': diff,
                'noise_scale': noise_scale,
                'severity': severity,
            })

    has_breakpoint = any(bp['severity'] == 'breakpoint' for bp in breakpoints)
    return breakpoints, not has_breakpoint
```

```python
def _changepoint_ci(x_left, x_right, p_hat, N_j, r, y,
                    alpha=0.5, gamma=0.05, n_boot=500,
                    theta_tilde=None, blocks=None, seed=None):
    """Bootstrap 变更点置信区间（在固定箱边界下）。

    重采样 (r_i, y_i) 对 → 固定边界上重新计数 → PAVA → ĵ*_b。
    ĵ* 的 (1−γ) 置信集 = 覆盖 (1−γ) 比例 ĵ*_b 的连续索引范围。

    Args:
        theta_tilde, blocks: 可选，预计算的 PAVA 输出。
            若调用方已在 _fit_vulnerability_pava 中跑过 PAVA，
            传入此参数可避免重复计算。
        seed: 可选，Bootstrap 随机种子。传入固定 int 可复现结果（测试必需）。

    Returns:
        dict with: j_star, j_star_lo, j_star_hi,
                   x_star_pt, x_star_lo, x_star_hi
    """
    # _pava 定义在同一模块中，直接调用，无需 import
    J = len(p_hat)
    n = len(r)
    rng = np.random.default_rng(seed)  # REVIEW-FIX-PREV: ISSUE-002 种子化保证可复现

    # 点估计——若调用方已传 PAVA 结果则直接复用
    if theta_tilde is None or blocks is None:
        theta_tilde, _, blocks = _pava(p_hat, N_j)
    theta_per_bin = np.zeros(J)
    for bi, blk in enumerate(blocks):
        for j in blk:
            theta_per_bin[j] = theta_tilde[bi]
    j_star = max((j for j in range(J) if theta_per_bin[j] > alpha), default=None)

    # Bootstrap
    j_star_boot = []
    for _ in range(n_boot):
        idx = rng.choice(n, size=n, replace=True)
        r_b, y_b = r[idx], y[idx]
        p_b = np.zeros(J)
        N_j_boot = np.zeros(J, dtype=int)
        for j in range(J):
            mask = (r_b >= x_left[j]) & (r_b < x_right[j])
            if j == J - 1:
                mask = (r_b >= x_left[j])
            nj = mask.sum()
            N_j_boot[j] = nj
            p_b[j] = y_b[mask].sum() / nj if nj > 0 else 0.0
        # nj=0 的箱权重置 1（避免零权重；p_b=0 无影响）
        N_j_boot[N_j_boot == 0] = 1
        theta_b, _, blocks_b = _pava(p_b, N_j_boot)  # REVIEW-FIX-PREV: ISSUE-012 使用 Bootstrap 样本量 N_j_boot 替代原始 N_j——权重与重采样数据匹配，避免变更点估计向原始样本箱结构偏移
        theta_per_b = np.zeros(J)
        for bi, blk in enumerate(blocks_b):
            for j in blk:
                theta_per_b[j] = theta_b[bi]
        js = max((j for j in range(J) if theta_per_b[j] > alpha), default=None)
        if js is not None:
            j_star_boot.append(js)

    if j_star_boot:
        j_star_lo = int(np.percentile(j_star_boot, 100 * gamma / 2))
        j_star_hi = int(np.percentile(j_star_boot, 100 * (1 - gamma / 2)))
    else:
        # ISSUE-FIX-033: 无 Bootstrap 变更点 → 全部返回 None（与 j_star 一致），
        #   避免 x_star_lo/hi 回退到 [0, x_right[0]] 与 x_star_pt=None 矛盾
        j_star_lo = j_star_hi = None

    return {
        'j_star': j_star,
        'j_star_lo': j_star_lo,
        'j_star_hi': j_star_hi,
        'x_star_pt': x_right[j_star] if j_star is not None else None,
        'x_star_lo': x_left[j_star_lo] if j_star_lo is not None else None,
        'x_star_hi': x_right[j_star_hi] if j_star_hi is not None else None,
    }
```
