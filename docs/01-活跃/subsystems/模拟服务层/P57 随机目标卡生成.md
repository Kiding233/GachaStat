# P57 随机目标卡生成

> 日期：2026-06-16 | 状态：设计中
> 触发：用户需求——按比例随机生成目标卡，每次模拟独立抽取，考察策略在任意目标组合下的表现
> 设计文档：`docs/03-归档/plans/2026-06-16-P57-随机目标卡生成-设计文档.md`

## 一、问题

当前目标卡完全依赖手动配置——用户在 TOML 或 GUI 中逐一指定 `[[targets]]`。这导致：

1. **策略评估有偏**：所有模拟共享同一套目标卡集合，无法回答「策略在任意目标组合下的平均表现」
2. **过度拟合风险**：用户可能下意识选择「好打」的目标卡，低估真实难度
3. **缺少随机性维度**：现有分析框架无法评估策略对目标卡组合的敏感度

## 二、目标

1. 支持按比例 a% 从 SSR 候选池中随机抽取目标卡，每轮模拟独立重新抽取
2. 手动配置的目标卡作为「固定底牌」，随机抽取时自动排除，保证不重合
3. 合并后目标卡总数 = ceil(a% × 候选池大小)，每张需求量固定为 1
4. 与现有 GDR 分析管线完全兼容——17 个指标逐条使用各自的 target_specs
5. CLI / GUI / 多进程 / 单进程 统一路径，零破坏性变更
6. 向后兼容——默认关闭，现有 7 个调用点无需修改

## 三、向后兼容策略

`run_batch_parallel` 保持签名不变，新增可选参数 `provider`：

```python
def run_batch_parallel(
    env: SimulationEnv,
    target_specs: Dict[str, int],        # 保持必选——所有现有调用点不变
    initial_resources: Dict[str, float],
    num_simulations: int,
    max_workers: int,
    seed: int = 0,
    progress_callback=None,
    strategy_name='',
    strategy_params=None,
    on_result=None,
    provider: 'TargetCardProvider' = None,  # 新增可选参数
) -> 'BatchResult':
```

- `provider=None`（默认）→ 内部用 `target_specs` 构建 `FixedProvider`——**与现有行为完全一致**
- `provider` 传入 → 使用 `provider.resolve(seed)` 生成每轮 `TargetCardSet`——`target_specs` 仅用于 GDRContext 兼容

**结果：7 个调用点零改动。**

```
# 位置：core/target_card.py，TargetCardSet 之后

import math
import random

class TargetCardProvider:
    """目标卡提供者——每次模拟根据 seed 返回确定性的 TargetCardSet。"""
    def resolve(self, seed: int) -> TargetCardSet:
        raise NotImplementedError


class FixedProvider(TargetCardProvider):
    """固定目标卡——始终返回同一套集合（现有行为）。"""
    def __init__(self, targets: TargetCardSet):
        self._targets = targets

    def resolve(self, seed: int) -> TargetCardSet:
        return self._targets


class RandomSampleProvider(TargetCardProvider):
    """从 SSR 候选池排除手动卡后，随机补足到 ceil(pct% × |候选池|)。

    合并规则：
    - 候选池 U = {全部SSR} 或 {限定SSR}
    - 手动卡 M = 用户在配置中指定的固定底牌
    - 随机池 = U \\ M（自动排除手动卡，不重合）
    - 目标总数 target_count = ceil(pct% × |U|)
    - 随机抽取 k = max(0, target_count - |M|) 张
    - 合并后总数 = max(|M|, target_count)
    - 每张 quantity_needed = 1
    """

    def __init__(
        self,
        candidate_pool: list[str],          # SSR 候选池
        manual_ids: set[str],               # 手动卡 ID（会被排除）
        pct: float,                         # 比例 0-100
        pool_lookup: dict[str, list[str]],  # card_id → [pool_ids]
    ):
        # 所有字段均为内置类型 → 可 pickle → 可传入 worker 子进程
        self._candidate_pool = list(candidate_pool)
        self._manual_ids = manual_ids
        self._target_count = max(1, math.ceil(len(candidate_pool) * pct / 100.0))
        self._pool_lookup = dict(pool_lookup)

    def resolve(self, seed: int) -> TargetCardSet:
        # 局部 Random 实例——不污染模块级 random 状态，保证确定性
        rng = random.Random(seed)

        # 计算需要随机补足的数量
        needed = max(0, self._target_count - len(self._manual_ids))

        # 候选池排除手动卡
        available = [cid for cid in self._candidate_pool
                     if cid not in self._manual_ids]

        # 随机抽取
        # 边界情形：
        #   - available 为空（无 SSR card_def）→ k=0 → 仅保留手动卡
        #   - needed ≤ 0（手动卡已超额：|M| ≥ ceil(pct% × |U|)）→ k=0 → 仅保留手动卡
        #   - normal case: k = min(needed, len(available))
        if needed <= 0 or len(available) == 0:
            k = 0
        else:
            k = min(needed, len(available))
        picked = rng.sample(available, k) if k > 0 else []

        targets: list[TargetCard] = []

        # 手动卡（固定底牌）
        for cid in sorted(self._manual_ids):
            targets.append(TargetCard(
                card_id=cid,
                pool_ids=self._pool_lookup.get(cid, []),
                quantity_needed=1,
            ))

        # 随机卡
        for cid in sorted(picked):
            targets.append(TargetCard(
                card_id=cid,
                pool_ids=self._pool_lookup.get(cid, []),
                quantity_needed=1,
            ))

        return TargetCardSet(targets)


def create_provider(store: 'ConfigStore') -> TargetCardProvider:
    """根据 ConfigStore 构造合适的 Provider。

    规则：
    - random_target_enabled=False → FixedProvider（现有行为）
    - random_target_enabled=True  → RandomSampleProvider（手动+随机叠加）
    """
    from gacha_simulator.core import TargetCard

    manual_ids = {tc.card_id for tc in store.target_cards}
    manual_targets = [
        TargetCard(card_id=tc.card_id, pool_ids=list(tc.pool_ids),
                   quantity_needed=tc.quantity)
        for tc in store.target_cards
    ]
    manual_set = TargetCardSet(manual_targets)

    if not getattr(store, 'random_target_enabled', False):
        return FixedProvider(manual_set)

    # 构建 SSR 候选池
    scope = getattr(store, 'random_target_scope', 'limited_ssr')
    ssr_ids = _collect_ssr_ids(store, scope)

    # 构建 pool_lookup: card_id → [pool_ids]
    pool_lookup = _build_pool_lookup(store)

    return RandomSampleProvider(
        candidate_pool=ssr_ids,
        manual_ids=manual_ids,
        pct=getattr(store, 'random_target_pct', 30.0),
        pool_lookup=pool_lookup,
    )


def _collect_ssr_ids(store: 'ConfigStore', scope: str) -> list[str]:
    """按 scope 收集 SSR 候选 ID 列表。"""
    if scope == 'all_ssr':
        return [cd.card_id for cd in store.card_defs
                if getattr(cd, 'rarity', '').upper() == 'SSR']
    else:  # 'limited_ssr' 或其他默认值
        # 限定 SSR = rarity='SSR' 且 id 以 'limited' 开头
        return [cd.card_id for cd in store.card_defs
                if getattr(cd, 'rarity', '').upper() == 'SSR'
                and cd.card_id.startswith('limited')]


def _build_pool_lookup(store: 'ConfigStore') -> dict[str, list[str]]:
    """从 ConfigStore 构建 card_id → [pool_ids] 映射。"""
    lookup: dict[str, list[str]] = {}
    for cd in store.card_defs:
        pools = list(getattr(cd, 'pools', []))
        if pools:
            lookup[cd.card_id] = pools
    # 兜底：没有 card_defs.pools 信息时从 TargetCardEntry 获取
    for tc in store.target_cards:
        if tc.card_id not in lookup and tc.pool_ids:
            lookup[tc.card_id] = list(tc.pool_ids)
    return lookup
```

```python
def test_fixed_provider_same_seed():
    """FixedProvider 无论 seed 如何都返回相同集合。"""
    ts = TargetCardSet([TargetCard("A", ["p1"], 1)])
    p = FixedProvider(ts)
    assert p.resolve(0) is ts
    assert p.resolve(42) is ts

def test_random_sample_deterministic():
    """同 seed 返回同结果。"""
    p = RandomSampleProvider(["A","B","C","D"], set(), 50.0, {})
    a = p.resolve(42)
    b = p.resolve(42)
    assert [t.card_id for t in a.targets] == [t.card_id for t in b.targets]

def test_random_sample_excludes_manual():
    """手动卡被排除，不重复。"""
    p = RandomSampleProvider(["A","B","C","D"], {"A","B"}, 50.0, {})
    result = p.resolve(0)
    ids = {t.card_id for t in result.targets}
    assert "A" in ids  # 手动保留
    assert "B" in ids  # 手动保留
    assert len(ids) == max(2, ceil(4*0.5))  # = max(2, 2) = 2

def test_random_sample_count():
    """合并后总数 = ceil(pct% × |U|)。"""
    p = RandomSampleProvider(["A","B","C","D","E","F"], set(), 50.0, {})
    result = p.resolve(0)
    assert len(result.targets) == 3  # ceil(6*0.5) = 3

def test_create_provider_disabled():
    """关闭随机 → FixedProvider。"""

    p = create_provider(store)
    assert isinstance(p, FixedProvider)

def test_create_provider_enabled():

    store.random_target_enabled = True
    store.random_target_pct = 30.0
```

```python
def _wk_run_single(args):
    """子进程 worker 入口——每轮 resolve 目标集 + 新建 extractor。"""
    seed, initial_resources = args

    # ① 获取本轮 TargetCardSet
    target_set = _wk_provider.resolve(seed)

    # ② 用本轮目标新建 WorkerLocalExtractor
    ts = {t.card_id: t.quantity_needed for t in target_set.targets}
    from gacha_simulator.core.streaming import WorkerLocalExtractor
    extractor = WorkerLocalExtractor(
        pool_end_times=_wk_env.pool_end_times,
        target_ids=set(ts.keys()),
        ssr_ids=_wk_env.ssr_ids,
        target_specs=ts,
        initial_resources=_wk_env.initial_resources,
        n_heatmap_bins=getattr(_wk_env, 'n_heatmap_bins', 50),
        max_keep=min(200, max(10, len(ts) * 2 + 10)),
    )

    # ③ 模拟
    try:
        compact = _run_single(_wk_env, target_set, seed, initial_resources)
    except Exception:
        traceback.print_exc()
        return (None, None) if _wk_return_compact else None

    if compact is None:
        return (None, None) if _wk_return_compact else None

    # ④ 提取
    extraction = None
    try:
        extraction = extractor.process(compact)
    except Exception:
        traceback.print_exc()

    if _wk_return_compact:
        return (compact, extraction)
    return extraction
```

## 六、风险与缓解

| 风险 | 等级 | 缓解 |
|------|------|------|
| `_wk_init` 签名变更 → Windows spawn pickle | 已消除 | provider 通过 `env.target_card_provider` 传递，`_wk_init` 签名/`initargs` 不变 |
| extractor 每轮新建 → 性能退化 | 极低 | 实测 ~10μs，模拟一次 ~10-100ms |
| GDR 逐条计算 → 分析面板变慢 | 极低 | 1000×17 次 checker ≈ 5ms |
| 旧 aggregate 数据无 `target_specs` | 低 | `agg.get('target_specs', ctx.target_specs)` 兜底 |
| 手动/随机目标重复 | 无 | 候选池显式排除 `manual_ids` |
| 候选池为空（无 SSR card_def） | 已消除 | `resolve()` 内显式 guard：`len(available)==0 → k=0` |
| `merge_extraction_packets` 丢弃 `target_specs` | 已消除 | `target_specs` 写入 `agg` 内部，`pkt.get('aggregate')` 自然透传 |
| no_draw 路径多余 provider 构建 | 已消除 | `no_draw` 调用不传 `provider` |
